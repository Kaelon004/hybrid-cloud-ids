import os
import re
import time
import socket
import json
import logging
import threading
import traceback
import psutil
from collections import defaultdict, deque
from datetime import datetime, timedelta
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

# Enhanced logging configuration
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('s_hids_agent.log'),
        logging.StreamHandler()
    ]
)

class HIDSAgent:
    def __init__(self, 
                 manager_host, 
                 manager_port, 
                 monitored_paths,
                 max_failed_attempts=5, 
                 time_window=300,  # 5 minutes
                 log_paths=['/var/log/auth.log']):
        self.manager_host = manager_host
        self.manager_port = manager_port
        self.max_failed_attempts = max_failed_attempts
        self.time_window = time_window
        self.log_paths = log_paths
        self.failed_attempts = defaultdict(list)
        self.monitored_paths = [os.path.abspath(path) for path in monitored_paths]
        self.event_queue = []
        self.queue_lock = threading.Lock()
        self.file_positions = {}
        self.throughput_window = deque(maxlen=10)
        self.resource_usage_log_interval = 10
        threading.Thread(target=self.monitor_resources, daemon=True).start()

        logging.info("HIDS Agent Initialized")

    def start(self):
        try:
            # Initialize file positions for log monitoring
            for log_path in self.log_paths:
                if os.path.exists(log_path):
                    with open(log_path, 'r') as log_file:
                        self.file_positions[log_path] = log_file.seek(0, 2)  # Start at end of file
                else:
                    logging.warning(f"Log file does not exist: {log_path}")
                    self.file_positions[log_path] = 0

            # Start threads for monitoring
            threading.Thread(target=self.monitor_system_logs, daemon=True).start()
            threading.Thread(target=self.send_events_periodically, daemon=True).start()

            # File monitoring setup
            observer = Observer()
            handler = FileIntegrityHandler(self.manager_host, self.manager_port, self.monitored_paths, self)

            for path in self.monitored_paths:
                if os.path.exists(path):
                    observer.schedule(handler, path=path, recursive=True)
                else:
                    logging.warning(f"Monitored path does not exist: {path}")

            observer.start()

            logging.info("HIDS Agent started successfully.")
            while True:
                time.sleep(10)
        except Exception as e:
            logging.error(f"Error during startup: {e}")
            logging.error(traceback.format_exc())
        finally:
            observer.stop()
            observer.join()

    def monitor_resources(self):
        while True:
            cpu_usage = psutil.cpu_percent(interval=1)
            memory_info = psutil.virtual_memory()
            logging.info(f"CPU Usage: {cpu_usage}%, Memory Usage: {memory_info.percent}%")
            time.sleep(self.resource_usage_log_interval)

    def monitor_system_logs(self):
        while True:
            try:
                current_time = datetime.now()
                for log_path in self.log_paths:
                    if not os.path.exists(log_path):
                        logging.warning(f"Log file not found: {log_path}")
                        continue

                    with open(log_path, 'r') as log_file:
                        log_file.seek(self.file_positions.get(log_path, 0))
                        while True:
                            line = log_file.readline()
                            if not line:
                                break
                            attempt = self.parse_login_attempt(line)
                            if attempt:
                                self.track_login_attempts(attempt['ip'], attempt['user'], current_time)

                        self.file_positions[log_path] = log_file.tell()
            except Exception as e:
                logging.error(f"Error monitoring logs: {e}")
            time.sleep(2)

    def parse_login_attempt(self, log_line):
        patterns = [
            r'Failed password for (?P<user>\w+) from (?P<ip>\d{1,3}(?:\.\d{1,3}){3})',
            r'Invalid user (?P<user>\w+) from (?P<ip>\d{1,3}(?:\.\d{1,3}){3})',
            r'sshd.*authentication failure.*rhost=(?P<ip>\d{1,3}(?:\.\d{1,3}){3})'
        ]
        for pattern in patterns:
            match = re.search(pattern, log_line, re.IGNORECASE)
            if match:
                return {
                    'ip': match.group('ip'),
                    'user': match.group('user') if 'user' in match.groupdict() else 'unknown'
                }
        return None

    def track_login_attempts(self, ip, user, current_time):
        attempt_key = f"{ip}_{user}"
        self.failed_attempts[attempt_key] = [
            attempt for attempt in self.failed_attempts[attempt_key] 
            if current_time - attempt < timedelta(seconds=self.time_window)
        ]
        self.failed_attempts[attempt_key].append(current_time)

        if len(self.failed_attempts[attempt_key]) >= self.max_failed_attempts:
            self.report_event("Brute Force Attack Detected", {
                "ip": ip,
                "user": user,
                "failed_attempts": len(self.failed_attempts[attempt_key])
            })

    def report_event(self, event_type, details, ground_truth=None):
        event = {
            "event": event_type,
            "details": details,
            "detected_time": datetime.now().strftime('%Y-%m-%d %H:%M:%S'),
            "ground_truth": ground_truth
        }
        with self.queue_lock:
            self.event_queue.append(event)
            self.throughput_window.append(datetime.now())

    def calculate_throughput(self):
        if len(self.throughput_window) < 2:
            return 0
        time_diff = (self.throughput_window[-1] - self.throughput_window[0]).total_seconds()
        return len(self.throughput_window) / time_diff if time_diff > 0 else 0

    def send_events_periodically(self):
        while True:
            try:
                with self.queue_lock:
                    events_to_send = self.event_queue.copy()
                    self.event_queue.clear()

                for event in events_to_send:
                    event["throughput"] = self.calculate_throughput()
                    self.send_event(event)
            except Exception as e:
                logging.error(f"Error sending events: {e}")
            time.sleep(2)

    def send_event(self, event):
        try:
            with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as client_socket:
                client_socket.connect((self.manager_host, self.manager_port))
                client_socket.send(json.dumps(event).encode('utf-8'))
                logging.info(f"Event sent: {event}")
        except Exception as e:
            logging.error(f"Failed to send event: {e}")

class FileIntegrityHandler(FileSystemEventHandler):
    def __init__(self, manager_host, manager_port, monitored_paths, agent):
        self.manager_host = manager_host
        self.manager_port = manager_port
        self.monitored_paths = monitored_paths
        self.agent = agent

    def is_path_monitored(self, filepath):
        filepath = os.path.abspath(filepath)
        return any(os.path.commonpath([filepath, path]) == path for path in self.monitored_paths)

    def process_event(self, event_type, filepath):
        if self.is_path_monitored(filepath):
            self.agent.report_event(f"File {event_type}", {"file": filepath})

    def on_created(self, event):
        self.process_event("Created", event.src_path)

    def on_deleted(self, event):
        self.process_event("Deleted", event.src_path)

    def on_modified(self, event):
        self.process_event("Modified", event.src_path)

    def on_moved(self, event):
        self.process_event("Moved", event.dest_path)
